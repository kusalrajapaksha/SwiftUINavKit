Navigation
Done Right

A powerful navigation router for SwiftUI that bridges UIKit's UINavigationController with SwiftUI, providing programmatic navigation with type-safety.

Get Started → View Docs 
struct HomeView: View {
    @EnvironmentObject var router: NavigationRouter
    
    var body: some View {
        VStack {
            Button("Go to Detail") {
                router.push(DetailView())
            }
            
            Button("Deep Link to Settings") {
                router.push(SettingsView(), routeID: "settings")
            }
        }
        .navigationTitle("Home")
    }
}
